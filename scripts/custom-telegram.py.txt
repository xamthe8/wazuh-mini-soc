#!/usr/bin/env python3
import sys
import json
import requests

alert_file = sys.argv[1]
chat_id = sys.argv[2]
hook_url = sys.argv[3]

with open(alert_file) as f:
    alert = json.load(f)

rule = alert.get("rule", {})
agent = alert.get("agent", {})
ev = alert.get("data", {}).get("win", {}).get("eventdata", {})
mitre = ", ".join(rule.get("mitre", {}).get("id", []))

lines = [
    "Wazuh alert (level {})".format(rule.get("level")),
    "Rule {}: {}".format(rule.get("id"), rule.get("description")),
    "Agent: {}".format(agent.get("name")),
    "Time (UTC): {}".format(alert.get("timestamp")),
]
if mitre:
    lines.append("MITRE: {}".format(mitre))
if ev.get("image"):
    lines.append("Image: {}".format(ev["image"]))
if ev.get("commandLine"):
    lines.append("Command: {}".format(ev["commandLine"][:300]))
if ev.get("user"):
    lines.append("User: {}".format(ev["user"]))

requests.post(hook_url, json={"chat_id": chat_id, "text": "\n".join(lines)}, timeout=10)
